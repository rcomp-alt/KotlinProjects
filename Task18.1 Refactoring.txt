data class Person(val name: String, val age: Int)

fun main() {
    val people = listOf(
        Person("Алексей", 25),
        Person("Мария", 30),
        Person("Иван", 17),
        Person("Ольга", 22)
    )
    
    val adults = people
        .filter { it.age >= 18 }
        .sortedBy { it.name }
        .map { it.name }
    
    println(adults)
}
